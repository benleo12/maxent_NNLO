"""Frozen generator-portability comparison using the released DY inputs."""
import gzip
import hashlib
import json
import time
from pathlib import Path

import numpy as np
from maxent_upgrade import (bin_diagnostics, load_moments, released,
                            upgrade, upgrade_with_uncertainties, weight_diagnostics)
from maxent_upgrade.provenance import json_value

ROOT=Path('/work')
OUT=ROOT/'analysis'
OUT.mkdir(exist_ok=False)
reference=dict(np.load(ROOT/'data/dy_reference.npz', allow_pickle=False))
outputs={}

def report_json(value):
    if isinstance(value,dict):
        return {str(k):report_json(v) for k,v in value.items()}
    if isinstance(value,np.ndarray):
        return report_json(value.tolist())
    if isinstance(value,(list,tuple)):
        return [report_json(v) for v in value]
    if isinstance(value,np.generic):
        return report_json(value.item())
    if isinstance(value,float) and not np.isfinite(value):
        return None
    return value

def normalized_reference(name):
    runs=reference[name+'_per_run']
    edges=reference[name+'_edges']
    omitted=[]
    if name=='angle':
        norm=(runs*np.diff(edges)[None,:,None]).sum(axis=1)
    else:
        with gzip.open(ROOT/'code/moments/dy_nnlo_moments_runs.json.gz','rt') as f:
            released_runs=json.load(f)
        lookup={s:i for i,s in enumerate(released_runs['runs'])}
        keep=np.array([int(s) in lookup for s in reference['recoil_seeds']])
        omitted=reference['recoil_seeds'][~keep].tolist()
        runs=runs[keep]
        assert keep.sum()>=2
        norm=np.array([[released.cross_section('dy',scale=k,runs=[lookup[int(s)]])
                        for k in range(7)] for s in reference['recoil_seeds'][keep]])
    means=runs.mean(axis=0)
    normalization=norm.mean(axis=0)
    shapes=means/normalization[None,:]
    influence=(runs[:,:,0]-shapes[None,:,0]*norm[:,None,0])/normalization[0]
    return dict(edges=edges, central=shapes[:,0], scale_members=shapes.T,
                sample_error=influence.std(axis=0,ddof=1)/np.sqrt(len(runs)),
                scale_low=shapes.min(axis=1),scale_high=shapes.max(axis=1),
                normalization_pb=normalization,independent_runs=len(runs),
                omitted_unpaired_seeds=np.array(omitted,dtype=int))

for name in ('angle','recoil'):
    np.savez_compressed(OUT/(name+'_reference.npz'),**normalized_reference(name))

parts=[]
provenance=[]
for shard in range(1,161):
    path=ROOT/f'runs/run_{shard}/events.npz'
    with np.load(path,allow_pickle=False) as data:
        parts.append({k:data[k] for k in data.files if data[k].ndim==1})
    provenance.append(json.loads(path.with_suffix('.json').read_text()))
herwig={k:np.concatenate([p[k] for p in parts]) for k in parts[0]}
assert len(np.unique(herwig['event_id']))==len(herwig['event_id'])
assert len(herwig['event_id'])>=500000
assert sum(p['source_records'] for p in provenance)==1600000
np.savez_compressed(OUT/'herwig_prior.npz',**herwig,
                    record_provenance_json=np.array(json.dumps(provenance,sort_keys=True)))
del parts

# Fixed public-sample recipe, declared before completing production or fitting.
rng=np.random.default_rng(731131)
stratum_edges=[0,30,60,100,200,500,np.inf]
caps=[50000,20000,20000,15000,12000,12000]
indices=[]
probabilities=[]
strata=[]
for low,high,cap in zip(stratum_edges[:-1],stratum_edges[1:],caps):
    eligible=np.flatnonzero((herwig['pT_ll']>=low)&(herwig['pT_ll']<high))
    number=min(len(eligible),cap)
    chosen=rng.choice(eligible,number,replace=False) if number else eligible
    probability=number/len(eligible) if len(eligible) else 0
    indices.extend(chosen)
    probabilities.extend([probability]*number)
    strata.append(dict(low=low,high=high if np.isfinite(high) else None,
                       population=len(eligible),selected=number,inclusion_probability=probability))
order=np.argsort(indices)
indices=np.array(indices,dtype=int)[order]
probabilities=np.array(probabilities)[order]
public={key:value[indices] for key,value in herwig.items()}
public['original_weight']=public['weight'].copy()
public['weight']/=probabilities
public['inclusion_probability']=probabilities
public['source_index']=indices
record=dict(generator='Herwig 7.3.0p1',source_events=len(herwig['weight']),seed=731131,
            strata=strata,sampling='uniform without replacement within recoil strata',
            weight='original_weight / inclusion_probability',
            unique_event_id='generator_event_number + shard_id * 10000000',
            generation_seeds='731101 + shard_id',generated_events_per_shard=10000,
            shards=160,protocol='PROTOCOL_HERWIG.txt',source_shards=provenance)
public['fixture_provenance_json']=np.array(json.dumps(record,sort_keys=True))
np.savez_compressed(OUT/'herwig_public_prior.npz',**public)
(OUT/'herwig_public_prior.json').write_text(json.dumps(record,indent=2)+'\n')
del public

with np.load(ROOT/'data/dy_full_prior.npz',allow_pickle=False) as data:
    pythia=dict(data)

for label,events in [('pythia',pythia),('herwig',herwig)]:
    start=time.monotonic()
    ensemble=upgrade_with_uncertainties(events,'dy',n_boot=20,seed=1,
        progress=lambda name,result: print(label,name,result.summary(),flush=True))
    elapsed=time.monotonic()-start
    input_path=OUT/'herwig_prior.npz' if label=='herwig' else ROOT/'data/dy_full_prior.npz'
    ensemble.provenance['input_file']=dict(name=input_path.name,
        sha256=hashlib.sha256(input_path.read_bytes()).hexdigest())
    np.savez_compressed(OUT/(label+'_weights.npz'),event_id=events['event_id'],**ensemble.arrays())
    moments,config=load_moments('dy')
    config['fixed_orders']={'mll':12,'y_abs':11,'pT_ll':16}
    common=upgrade(events,moments,config)
    np.savez_compressed(OUT/(label+'_common_weights.npz'),event_id=events['event_id'],weights=common.weights)
    report=dict(events=len(events['weight']),elapsed_seconds=elapsed,
                cross_section_pb=ensemble.cross_section,orders=ensemble.central.chosen_moments,
                common_orders=common.chosen_moments,
                statistics=weight_diagnostics(ensemble.central.weights,events['weight']),
                notes=ensemble.notes,fit_report=ensemble.central.report,
                common_fit_report=common.report,local_statistics={})
    variables=[('recoil','pT_ll',reference['recoil_edges']),
               ('full_recoil','pT_ll',np.r_[.5,1,2,3,4,5,7,10,15,20,25,30,40,50,60,80,100,140,200,300,500]),
               ('angle','abs_costh',reference['angle_edges']),
               ('mass','mll',np.linspace(66,116,26)),
               ('rapidity','y_abs',np.linspace(0,2.5,26)),
               ('leading_lepton','pT_lead',np.array([27,30,35,40,45,50,60,80,110,150,250,500])),
               ('subleading_lepton','pT_sub',np.array([27,30,35,40,45,50,60,80,110,150,250])),
               ('phistar','phistar',np.r_[0,np.geomspace(.001,3,25)])]
    prior=events['weight']/events['weight'].sum()
    for name,observable,edges in variables:
        hist=ensemble.histogram(events[observable],edges,unit_area=True)
        widths=np.diff(edges)
        for source,weights in ensemble.weights.items():
            hist[source+'_members']=np.array([np.histogram(events[observable],edges,weights=w)[0]/widths
                                              for w in weights])
        hist['prior']=np.histogram(events[observable],edges,weights=prior)[0]/widths
        hist['common']=np.histogram(events[observable],edges,weights=common.weights)[0]/widths
        h=hist['prior']*widths
        h2=np.histogram(events[observable],edges,weights=prior**2)[0]
        hist['prior_sample_error']=np.sqrt(np.maximum((1-2*h)*h2+h*h*np.sum(prior**2),0))/widths
        np.savez_compressed(OUT/(label+'_'+name+'.npz'),**hist)
        report['local_statistics'][name]=bin_diagnostics(events[observable],edges,ensemble.central.weights)
    (OUT/(label+'_report.json')).write_text(json.dumps(report_json(report),indent=2,allow_nan=False)+'\n')
    outputs[label]={k:report[k] for k in ('events','elapsed_seconds','orders','common_orders','statistics')}
    del ensemble,common
(OUT/'summary.json').write_text(json.dumps(json_value(outputs),indent=2)+'\n')
print(json.dumps(json_value(outputs)),flush=True)
