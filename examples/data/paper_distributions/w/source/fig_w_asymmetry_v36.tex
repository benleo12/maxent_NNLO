\newcommand{\envelope}[3]{%
  \addplot[const plot,name path=bandlo,draw=none] table[x=edge,y=#2lo]{w_asymmetry_#1.dat};
  \addplot[const plot,name path=bandhi,draw=none] table[x=edge,y=#2hi]{w_asymmetry_#1.dat};
  \addplot[fill=#3!16,draw=none] fill between[of=bandlo and bandhi];}
\newcommand{\predictions}[2]{%
  \envelope{#1}{basic#2}{basic}\envelope{#1}{spin#2}{spin}
  \addplot[const plot,basic,densely dashed,line width=.9pt] table[x=edge,y=basic#2]{w_asymmetry_#1.dat};
  \addplot[const plot,spin,line width=.95pt] table[x=edge,y=spin#2]{w_asymmetry_#1.dat};}
\newcommand{\points}[3]{%
  \addplot[black,error,#3]
   table[x=center,y=#2,y error minus=refminus,y error plus=refplus]{w_asymmetry_#1.dat};}
\newcommand{\residuals}[2]{%
  \draw[black!50,line width=.45pt] (axis cs:0,0)--(axis cs:2.5,0);
  \predictions{#1}{d}
  \addplot[basic,error,mark=none] table[x expr=\thisrow{center}-.014,y=basicd,y error=basicerr]{w_asymmetry_#1.dat};
  \addplot[spin,error,mark=none] table[x expr=\thisrow{center}+.014,y=spind,y error=spinerr]{w_asymmetry_#1.dat};
  \points{#1}{zero}{#2}}
