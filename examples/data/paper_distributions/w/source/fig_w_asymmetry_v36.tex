\newcommand{\envelope}[3]{%
  \addplot[const plot,name path=bandlo,draw=none] table[x=edge,y=#2lo]{w_asymmetry_#1.dat};
  \addplot[const plot,name path=bandhi,draw=none] table[x=edge,y=#2hi]{w_asymmetry_#1.dat};
  \addplot[fill=#3!16,draw=none] fill between[of=bandlo and bandhi];}
\newcommand{\predictions}[2]{%
  \envelope{#1}{basic#2}{basic}\envelope{#1}{spin#2}{spin}
  \addplot[const plot,basic,densely dashed,line width=.9pt] table[x=edge,y=basic#2]{w_asymmetry_#1.dat};
  \addplot[const plot,spin,line width=.95pt] table[x=edge,y=spin#2]{w_asymmetry_#1.dat};}
\newcommand{\reference}[1]{%
  \addplot[black,error,nnlopoint]
    table[x expr=\thisrow{center}+.014,y=reference,y error=referr]{w_asymmetry_#1.dat};}
\newcommand{\ratios}[2]{%
  \draw[black!50,line width=.45pt] (axis cs:0,1)--(axis cs:2.5,1);
  \predictions{#1}{#2}
  \addplot[basic,error,mark=none] table[x expr=\thisrow{center}-.025,y=basic#2,y error=basic#2err]{w_asymmetry_#1.dat};
  \addplot[spin,error,mark=none] table[x expr=\thisrow{center}+.025,y=spin#2,y error=spin#2err]{w_asymmetry_#1.dat};}
\newcommand{\foratio}[1]{%
  \ratios{#1}{fo}
  \addplot[black,error,mark=none] table[x=center,y=unit,y error=ferr]{w_asymmetry_#1.dat};}
